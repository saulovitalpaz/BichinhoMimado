import React from 'react';
import FullCalendar from '@fullcalendar/react';
import timeGridPlugin from '@fullcalendar/timegrid';
import interactionPlugin from '@fullcalendar/interaction';
import ptBrLocale from '@fullcalendar/core/locales/pt-br';
import dayjs from 'dayjs';

const CalendarService = () => {
  const handleDateClick = (arg: any) => {
    alert('Novo agendamento em: ' + dayjs(arg.date).format('DD/MM/YYYY HH:mm'));
    // Aqui o Antigravity deve abrir o modal de Novo Agendamento
  };

  return (
    <div className="p-4 bg-white rounded-xl shadow-md">
      <FullCalendar
        plugins={[timeGridPlugin, interactionPlugin]}
        initialView="timeGridDay" // Visão diária é melhor para clínicas
        headerToolbar={{
          left: 'prev,next today',
          center: 'title',
          right: 'timeGridDay,timeGridWeek'
        }}
        locale={ptBrLocale}
        slotMinTime="07:00:00"
        slotMaxTime="20:00:00"
        editable={true}
        selectable={true}
        dateClick={handleDateClick}
        events={[
          { title: 'Bento (Banho) - João Silva', start: new Date(), color: '#2D6A4F' }
        ]}
      />
    </div>
  );
};
